// 針對 Safari / iOS 優化的 Base64 POST 請求 (避開 CORS Preflight 阻擋) async function captureAndDetect() { const scoreDisplay = document.getElementById('scoreDisplay'); const video = document.getElementById('cameraStream'); const canvas = document.getElementById('snapshotCanvas'); if (!video.videoWidth) { scoreDisplay.innerHTML = "⚠️ 相機尚未準備好，請稍候再試。"; return; } const maxDim = 1024; let targetWidth = video.videoWidth; let targetHeight = video.videoHeight; if (targetWidth > maxDim) { targetHeight = Math.round((targetHeight * maxDim) / targetWidth); targetWidth = maxDim; } canvas.width = targetWidth; canvas.height = targetHeight; const ctx2d = canvas.getContext('2d'); ctx2d.drawImage(video, 0, 0, targetWidth, targetHeight); scoreDisplay.innerHTML = "🔄 正在分析手牌..."; try { const dataUrl = canvas.toDataURL('image/jpeg', 0.75); const base64Data = dataUrl.split(',')[1]; // 使用原本運作成功的 Publishable Key 搭配標準 Inference 端點 const apiUrl = `https://detect.roboflow.com/${MODEL_ID}/${VERSION}?api_key=${API_KEY}&confidence=30`; const response = await fetch(apiUrl, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: base64Data }); if (!response.ok) { throw new Error(`HTTP 錯誤碼 (${response.status})`); } const data = await response.json(); const predictions = data.predictions || []; if (predictions.length === 0) { scoreDisplay.innerHTML = "⚠️ 未能辨識出任何麻雀牌，請調整距離或角度重試。"; return; } const rawClasses = predictions.map(p => p.class); const tiles14 = []; const unmapped = []; rawClasses.forEach(raw => { const norm = normalizeTileClass(raw); if (norm) tiles14.push(norm); else unmapped.push(raw); }); renderScore(tiles14, unmapped); } catch (err) { console.error("Detect Error:", err); scoreDisplay.innerHTML = `❌ 請求失敗：${err.message || '連線失敗'}`; } }
