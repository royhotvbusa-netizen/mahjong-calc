function captureFromFullscreen() { const scoreDisplay = document.getElementById('scoreDisplay'); // 優先抓全螢幕影片，若 Modal 未開啟則降級抓主畫面影片 const modal = document.getElementById('fullscreenCameraModal'); const isModalOpen = modal && modal.style.display !== 'none'; const video = isModalOpen ? document.getElementById('fullscreenVideo') : document.getElementById('cameraStream'); const scanBox = document.getElementById('scanBox'); const canvas = document.getElementById('snapshotCanvas'); if (!video || !video.videoWidth) { scoreDisplay.innerHTML = "⚠️ 相機尚未準備好，請稍候再試。"; if (isModalOpen) closeFullscreenCamera(); return; } scoreDisplay.innerHTML = "🔄 正在分析手牌..."; if (isModalOpen) closeFullscreenCamera(); const videoWidth = video.videoWidth; const videoHeight = video.videoHeight; let cropX = 0, cropY = 0, cropWidth = videoWidth, cropHeight = videoHeight; // 只有在全螢幕 Modal 開啟時，才進行框線精準裁切 if (isModalOpen && scanBox) { const videoRect = video.getBoundingClientRect(); const scanBoxRect = scanBox.getBoundingClientRect(); if (videoRect.width > 0 && videoRect.height > 0) { const scaleX = videoWidth / videoRect.width; const scaleY = videoHeight / videoRect.height; cropX = Math.max(0, (scanBoxRect.left - videoRect.left) * scaleX); cropY = Math.max(0, (scanBoxRect.top - videoRect.top) * scaleY); cropWidth = Math.min(videoWidth - cropX, scanBoxRect.width * scaleX); cropHeight = Math.min(videoHeight - cropY, scanBoxRect.height * scaleY); } } // 防呆機制：若計算出來的長寬無效，強制取全畫面 if (cropWidth <= 0 || cropHeight <= 0) { cropX = 0; cropY = 0; cropWidth = videoWidth; cropHeight = videoHeight; } canvas.width = Math.max(1, cropWidth); canvas.height = Math.max(1, cropHeight); const ctx2d = canvas.getContext('2d'); ctx2d.drawImage(video, cropX, cropY, cropWidth, cropHeight, 0, 0, canvas.width, canvas.height); canvas.toBlob((blob) => { ctx2d.clearRect(0, 0, canvas.width, canvas.height); canvas.width = 1; canvas.height = 1; if (!blob) { scoreDisplay.innerHTML = "❌ 圖片處理失敗，請重試。"; return; } const formData = new FormData(); formData.append("file", blob, "image.jpg"); const xhr = new XMLHttpRequest(); const apiUrl = `https://detect.roboflow.com/${MODEL_ID}/${VERSION}?api_key=${API_KEY}&confidence=30`; xhr.open("POST", apiUrl, true); xhr.onload = function () { if (xhr.status >= 200 && xhr.status < 300) { try { const data = JSON.parse(xhr.responseText); const predictions = data.predictions || []; if (predictions.length === 0) { scoreDisplay.innerHTML = "⚠️ 未能辨識出任何麻雀牌，請將手牌精準對準框線重試。"; return; } const rawClasses = predictions.map(p => p.class); const tiles14 = []; const unmapped = []; rawClasses.forEach(raw => { const norm = normalizeTileClass(raw); if (norm) tiles14.push(norm); else unmapped.push(raw); }); renderScore(tiles14, unmapped); } catch (e) { scoreDisplay.innerHTML = `❌ 解析回應失敗：${e.message}`; } } else { scoreDisplay.innerHTML = `❌ 伺服器錯誤 (HTTP ${xhr.status}): ${xhr.responseText || xhr.statusText}`; } }; xhr.onerror = function () { scoreDisplay.innerHTML = `❌ 網路連線失敗。請確認 API Key 配額或網路設定。`; }; xhr.send(formData); }, 'image/jpeg', 0.85); }
